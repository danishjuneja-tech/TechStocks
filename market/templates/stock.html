<script>


const symbol = "{{ symbol }}";


const apiUrl =

    "{% url 'stock_chart' symbol %}";



const ctx =

    document

        .getElementById("stockChart")

        .getContext("2d");



const chart = new Chart(

    ctx,

    {

        type: "line",


        data: {

            labels: [],


            datasets: [

                {

                    label: symbol + " Price",


                    data: [],


                    borderWidth: 3,


                    pointRadius: 2,


                    tension: 0.25,


                    fill: false

                }

            ]

        },


        options: {


            responsive: true,


            maintainAspectRatio: false,


            animation: false,


            scales: {


                x: {

                    title: {

                        display: true,

                        text: "Time"

                    }

                },


                y: {

                    title: {

                        display: true,

                        text: "Price (₹)"

                    }

                }


            }


        }

    }

);



async function updateChart() {


    try {


        const response =

            await fetch(

                apiUrl + "?t=" + Date.now()

            );


        const result =

            await response.json();



        document

            .getElementById("stockName")

            .innerText =

                result.name;



        document

            .getElementById("stockSymbol")

            .innerText =

                "(" + result.symbol + ")";



        document

            .getElementById("currentPrice")

            .innerText =

                "₹" +

                Number(

                    result.current_price

                ).toFixed(2);



        chart.data.labels =

            result.data.map(

                item => item.time

            );



        chart.data.datasets[0].data =

            result.data.map(

                item => item.price

            );



        chart.data.datasets[0].label =

            result.symbol + " Price";



        chart.update();


    }


    catch (error) {


        console.error(

            "Chart update failed:",

            error

        );


    }


}



updateChart();



setInterval(

    updateChart,

    2000

);
